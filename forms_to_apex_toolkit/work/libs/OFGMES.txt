

PACKAGE CG$MESSAGE IS

  /*
   * Call this function from the procedure defined in the 
   * MSGSFT preference.  'msg' will be returned containing
   * a string for display if it is a known generator message
   * otherwise it will be null.
   */
  procedure get_message(msgno  in     varchar2
                       ,msg    in out varchar2
                       ,param1 in     varchar2 default ''
                       ,param2 in     varchar2 default ''
                       ,param3 in     varchar2 default ''
                       ,param4 in     varchar2 default '');

  procedure split_string(msg    in     varchar2
                        ,pcount in     number
                        ,s1     in out varchar2
                        ,s2     in out varchar2
                        ,s3     in out varchar2
                        ,s4     in out varchar2
                        ,s5     in out varchar2);
 
  procedure mess_list(msgno  in  varchar2
                     ,str    out varchar2);


END;


PACKAGE BODY cg$message IS


procedure get_message
(msgno  in     varchar2
,msg    in out varchar2
,param1 in     varchar2 default ''
,param2 in     varchar2 default ''
,param3 in     varchar2 default ''
,param4 in     varchar2 default '')
IS
  tmp_string     varchar2(200) := msg;
  tmp_string1    varchar2(200) := '';
  tmp_string2    varchar2(200) := '';
  tmp_string3    varchar2(200) := '';
  tmp_string4    varchar2(200) := '';
  tmp_end_string varchar2(200) := '';
  param_count    number        := 4;
BEGIN

  mess_list(msgno, msg);

  -- find out how many parameters there are...
  if (param4 is null) then param_count := 3; end if;
  if (param3 is null) then param_count := 2; end if;
  if (param2 is null) then param_count := 1; end if;
  if (param1 is null) then param_count := 0; end if;

  if (param_count != 0)
  then
    split_string(msg,param_count,tmp_string1,tmp_string2,
                   tmp_string3,tmp_string4,tmp_end_string);

    msg := tmp_string1||param1||tmp_string2||param2||
           tmp_string3||param3||tmp_string4||param4||tmp_end_string;

  end if;

END; /* get_message */


PROCEDURE split_string
(msg    in     varchar2
,pcount in     number
,s1     in out varchar2
,s2     in out varchar2
,s3     in out varchar2
,s4     in out varchar2
,s5     in out varchar2)
IS
  tmp varchar2(200) := msg;
BEGIN

  -- chop up the message string
  if (pcount >= 1) then
    s1  := substr(tmp,1,instr(tmp,'%s')-1);
    tmp := substr(tmp,length(s1||'%s')+1);
  end if;

  if (pcount >= 2) then
    s2  := substr(tmp,1,instr(tmp,'%s')-1);
    tmp := substr(tmp,length(s2||'%s')+1);
  end if;

  if (pcount >= 3) then
    s3  := substr(tmp,1,instr(tmp,'%s')-1);
    tmp := substr(tmp,length(s3||'%s')+1);
  end if;

  if (pcount = 4) then
    s4  := substr(tmp,1,instr(tmp,'%s')-1);
    tmp := substr(tmp,length(s4||'%s')+1);
  end if;

  s5  := tmp;

END; /* split_string */


procedure mess_list
(msgno  in  varchar2
,str    out varchar2)
IS
BEGIN

  if (msgno =  '3') then 
    str := 'No row in table %s';
  elsif (msgno = '4') then 
    str := 'Cannot update %s while dependent %s exists';
  elsif (msgno =  '5') then 
    str := 'Cannot delete %s while dependent %s exists';
  elsif (msgno =  '6') then 
    str := 'Row exists already with same %s';
  elsif (msgno = '10') then 
    str := 'Invalid value for %s';
  elsif (msgno = '13') then 
    str := 'This %s does not exist';
  elsif (msgno = '14') then 
    str := 'Invalid combination of %s';
  elsif (msgno = '16') then 
    str := 'Call to procedure ''%s'' failed';
  elsif (msgno = '17') then 
    str := 'A reason for record deletion must be entered';
  elsif (msgno = '18') then 
    str := 'A reason for record creation must be entered';
  elsif (msgno = '19') then 
    str := 'A reason for record modification must be entered';
  elsif (msgno = '20') then 
    str := 'Derived item %s not updated';
  elsif (msgno = '21') then 
    str := 'No primary key row found for value in %s';
  elsif (msgno = '22') then 
    str := 'Press [Delete Record] again to confirm delete';
  elsif (msgno = '23') then 
    str := 'Help file is missing or incorrect';
  elsif (msgno = '24') then 
    str := 'More than one row in table %s';
  elsif (msgno = '29') then 
    str := 'More than one code controls row for given domain';
  elsif (msgno = '30') then 
    str := 'Key not valid in this context';
  elsif (msgno = '30') then 
    str := 'Key not valid in this context';
  elsif (msgno = '33') then 
    str := 'Query too complex - simplify and re-query';
  elsif (msgno = '35') then 
    str := 'Query of %s must be in context of %s';
  elsif (msgno = '36') then 
    str := 'Insert of %s must be in context of %s';
  elsif (msgno = '37') then 
    str := 'Query not allowed in this block';
  elsif (msgno = '49') then 
    str := 'More than one existing row with same key value';
  elsif (msgno = '52') then 
    str := 'Only one of %s may be entered';
  elsif (msgno = '53') then 
    str := 'One of %s required';
  elsif (msgno = '56') then 
    str := 'Summary data not accurate on restricted queries';
  elsif (msgno = '61') then 
    str := 'More than one primary key row for foreign key value';
  elsif (msgno = '62') then 
    str := 'Cleared row no longer exists; requery to display current rows';
  elsif (msgno = '63') then 
    str := 'Unable to call list of values form %s';
  elsif (msgno = '67') then 
    str := 'This form has outstanding changes to commit.';
  elsif (msgno = '68') then 
    str := 'Unable to call module %s';
  elsif (msgno = '71') then 
    str := 'Item is protected against update';
  elsif (msgno = '78') then 
    str := '%s must be entered';
  elsif (msgno = '82') then 
    str := '%s on %s caused the following violation:';
  elsif (msgno = '83') then 
    str := 'Validation failed on %s';
  elsif (msgno = '84') then 
    str := 'Validation failed on constraint %s';
  else
    str := null;
  end if;

END; /* mess_list */

END; /* cg$message */

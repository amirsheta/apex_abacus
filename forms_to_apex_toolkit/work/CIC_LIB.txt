PACKAGE BODY DCLTN_FN IS


	CURSOR DCLTN_FN_CD(V_DCLTN_TYPE NUMBER, V_DCLTN_STATUS NUMBER, V_TASK_ID NUMBER, V_DCLTN_LEVEL NUMBER, V_PORT_TYPE NUMBER) IS
	  			SELECT DISTINCT CD_FN_ID, EXEC_CODE
	  			FROM CODE_FNS CF, BUSINESS_FUNCTIONS BF
	  			WHERE BF.BUS_FN_ID = CF.BUS_FN_ID
	  			AND		BF.FN_EXE_TYPE = 1		
	  			AND		BF.REC_STATUS = 1
	  			AND		(CF.TASK_ID IS NULL OR CF.TASK_ID = V_TASK_ID)
	  			AND		(CF.DCLTN_TYPE IS NULL OR CF.DCLTN_TYPE = V_DCLTN_TYPE)	  			
	  			AND		(CF.STA_CD IS NULL OR CF.STA_CD = V_DCLTN_STATUS)
	  			AND		(CF.PORT_TYPE_ID IS NULL OR CF.PORT_TYPE_ID = V_PORT_TYPE)
	  			AND		(CF.BUS_LEVEL_ID IS NULL OR CF.BUS_LEVEL_ID = V_DCLTN_LEVEL)	  				  				  			
	  			AND		CF.REC_STATUS = 1
	  			;

---------------------------------------------------------------------------------------	
	PROCEDURE ADJ_DCLTN_FUNS(V_DCLTN_TYPE NUMBER, V_DCLTN_STATUS NUMBER, V_TASK_ID NUMBER, V_DCLTN_LEVEL NUMBER, V_PORT_TYPE NUMBER, V_PMT_METHOD NUMBER DEFAULT  1) IS
---------------------------------------------------------------------------------------	
	RES BOOLEAN;
	V_BANK_OFF NUMBER;
	HDATE DATE;
	V_MSG_TXT VARCHAR2(2000);
	V_PORT_CD NUMBER;
	V_DCLTN_ISN NUMBER;
	V_DCLTN_STA NUMBER;
	BEGIN
		V_PORT_CD := TO_NUMBER(NAME_IN('PARAMETER.P_DCLTN_PORT_CD'));
		V_DCLTN_ISN := TO_NUMBER(NAME_IN('PARAMETER.P_DCLTN_ISN'));
		
		IF NVL(V_PORT_CD, 0) <> 0  AND NVL(V_DCLTN_ISN, 0) <> 0 AND V_DCLTN_TYPE IN (1,2,3,4,5) THEN
			BEGIN
				SELECT 	DCLTN_STA_CD
				INTO		V_DCLTN_STA
				FROM 		DECLARATIONS
				WHERE 	PORT_CD = V_PORT_CD
				AND			DCLTN_ISN = V_DCLTN_ISN
				;


				IF ( ( (V_DCLTN_TYPE IN (1,2))   AND (V_DCLTN_STA = 17) ) OR
					   ( (V_DCLTN_TYPE IN (3,4,5)) AND (V_DCLTN_STA = 10) )
					 )  THEN
					TMPLT.DISABLE_ALL_FUNS;
			  	TMPLT.ENB_DIS_FUN_BUT ;
					RETURN;
				END IF;
				
			EXCEPTION
				WHEN OTHERS THEN
					NULL;
			END;
		END IF;
		
		HDATE := HIJDATE_DB.Cur_Hij_Date(V_MSG_TXT);
		IF HDATE IS NULL THEN
			TMPLT.DISABLE_ALL_FUNS;
			STR.DISPALERT(V_MSG_TXT, 'ERROR');
			RAISE FORM_TRIGGER_FAILURE;
		END IF;

		V_BANK_OFF := REVENUE.ALL_BANKS_OFF(TO_NUMBER(NAME_IN('PARAMETER.P_CURR_PORT_CD')), HDATE);
		TMPLT.DISABLE_ALL_FUNS;
		
		FOR I IN DCLTN_FN_CD(V_DCLTN_TYPE , V_DCLTN_STATUS , V_TASK_ID , V_DCLTN_LEVEL , V_PORT_TYPE ) LOOP
			IF LNK.GET_ALL_CONDS(I.CD_FN_ID) THEN
				IF V_PMT_METHOD = 2 THEN
					IF I.EXEC_CODE <> 'CD_00655' THEN
				  		TMPLT.ENABLE_FUN(I.EXEC_CODE);
					END IF;
				ELSE
					IF I.EXEC_CODE = 'CD_00650' THEN
						IF V_BANK_OFF = 0 THEN
							TMPLT.ENABLE_FUN('CD_00650');
						END IF;
					ELSIF I.EXEC_CODE = 'CD_00655' THEN
							IF V_BANK_OFF = 1 THEN
								TMPLT.ENABLE_FUN('CD_00655');
							END IF;
					ELSE				
				  		TMPLT.ENABLE_FUN(I.EXEC_CODE);
					END IF;
				END IF;
			END IF;
		END LOOP;			

  	TMPLT.ENB_DIS_FUN_BUT ;

  END ADJ_DCLTN_FUNS;


------------------------------------------------------------------------------
FUNCTION EXEC_DCLTN_FUNS(V_PORT_TYPE NUMBER, V_EXEC_CODE VARCHAR2, V_PORT_CD NUMBER, V_DCLTN_ISN NUMBER, 
													CALC_DT DATE, V_PAYER VARCHAR2, V_USER_ID VARCHAR2, V_TERMINAL_ID VARCHAR2,V_CRT_STRG BOOLEAN, V_CL_VAL OUT NUMBER) RETURN NUMBER IS
-----------------------------------------------------------------------------
	V_CRT_STRG_FLAG   NUMBER;
	V_BUS_FN_ID       NUMBER;
	V_CHK_FEES_RECALC NUMBER;
	NEED_COMMIT NUMBER := 0;
	V_MSG_TXT   VARCHAR2(2000);
	V_MSG_LEVEL NUMBER;
BEGIN
	IF V_CRT_STRG = TRUE THEN
		V_CRT_STRG_FLAG := 1;
	ELSE
		V_CRT_STRG_FLAG := 0;
	END IF;
	BEGIN
		SELECT DISTINCT CHK_FEES_RECALC
		INTO V_CHK_FEES_RECALC 
		FROM CODE_FNS CF, BUSINESS_FUNCTIONS BF
		WHERE CF.BUS_FN_ID = BF.BUS_FN_ID
		AND	CF.EXEC_CODE = V_EXEC_CODE
		;

	EXCEPTION
			WHEN OTHERS THEN
					STR.DISPALERT('INTERNAL ERROR IN CONTROL TABLE (CHECK RECALCULATE FEES) FOR CODE : ' || V_EXEC_CODE, 'ERROR');
					RAISE FORM_TRIGGER_FAILURE;
	END;

	IF V_CHK_FEES_RECALC = 1 THEN
		V_MSG_LEVEL := TO_NUMBER(NAME_IN('SYSTEM.MESSAGE_LEVEL'));
		COPY(25, 'SYSTEM.MESSAGE_LEVEL');
		NEED_COMMIT := DUTY.CALC_DCLTN_FEES (V_PORT_TYPE, V_PORT_CD, V_DCLTN_ISN, CALC_DT, V_PAYER, V_USER_ID, V_TERMINAL_ID, NULL, V_CRT_STRG_FLAG, V_CL_VAL, V_MSG_TXT);
		COPY(V_MSG_LEVEL, 'SYSTEM.MESSAGE_LEVEL');
		IF NEED_COMMIT = -1 THEN
			STR.DISPALERT(V_MSG_TXT, 'ERROR');
			RAISE FORM_TRIGGER_FAILURE;
		END IF;
	END IF;  -- V_CHK_FEES_RECALC = 1
	
	RETURN NEED_COMMIT;

END EXEC_DCLTN_FUNS;


---------------------------------------------------------------------------------------
	FUNCTION DCLTN_DISP_STATUS(V_DCLTN_TYPE NUMBER, V_DCLTN_STATUS NUMBER, V_TASK_ID NUMBER, V_DCLTN_LEVEL NUMBER, V_PORT_TYPE NUMBER) RETURN BOOLEAN IS
---------------------------------------------------------------------------------------	
	V_COUNT NUMBER := 0;
	V_PORT_CD NUMBER;
	V_DCLTN_ISN NUMBER;
	V_DCLTN_STA NUMBER;

	BEGIN
		V_PORT_CD := TO_NUMBER(NAME_IN('PARAMETER.P_DCLTN_PORT_CD'));
		V_DCLTN_ISN := TO_NUMBER(NAME_IN('PARAMETER.P_DCLTN_ISN'));
		
		IF NVL(V_PORT_CD, 0) <> 0  AND NVL(V_DCLTN_ISN, 0) <> 0 AND V_DCLTN_TYPE IN (1,2,3,4,5) THEN
			BEGIN
				SELECT 	DCLTN_STA_CD
				INTO		V_DCLTN_STA
				FROM 		DECLARATIONS
				WHERE 	PORT_CD = V_PORT_CD
				AND			DCLTN_ISN = V_DCLTN_ISN
				;


				IF ( ( (V_DCLTN_TYPE IN (1,2))   AND (V_DCLTN_STA = 17) ) OR
					   ( (V_DCLTN_TYPE IN (3,4,5)) AND (V_DCLTN_STA = 10) )
					 )  THEN
					RETURN FALSE;
				END IF;
				
			EXCEPTION
				WHEN OTHERS THEN
					NULL;
			END;
		END IF;

		FOR I IN DCLTN_FN_CD(V_DCLTN_TYPE , V_DCLTN_STATUS , V_TASK_ID , V_DCLTN_LEVEL , V_PORT_TYPE ) LOOP
			IF LNK.GET_ALL_CONDS(I.CD_FN_ID) THEN
				V_COUNT := V_COUNT + 1;
				EXIT;
			END IF;
		END LOOP;			
	  			
		IF V_COUNT <> 0 THEN
			RETURN TRUE;
		ELSE
			RETURN FALSE;
		END IF;						
	END DCLTN_DISP_STATUS;

END DCLTN_FN;

			Set_Block_Property(blk_id,INSERT_ALLOWED,PROPERTY_FALSE);  
    END IF;
		IF Updatabl = true THEN
     	Set_Block_Property(blk_id,UPDATE_ALLOWED,PROPERTY_TRUE);  
		ELSE
			Set_Block_Property(blk_id,UPDATE_ALLOWED,PROPERTY_FALSE);  
    END IF;     	
		IF Deletable = true THEN
     	Set_Block_Property(blk_id,DELETE_ALLOWED,PROPERTY_TRUE);  
		ELSE
			Set_Block_Property(blk_id,DELETE_ALLOWED,PROPERTY_FALSE);  			
    END IF;     	
   
   ELSE  
     STR.DISPALERT('Block '||blk_name||' does not exist.', 'ERROR');  
     RAISE Form_Trigger_Failure;  
   END IF;  
 END Make_Block_Full_Operation;

	PROCEDURE SET_BLOCK_ITEMS(cur_blk VARCHAR2,PROP NUMBER,PROP_VALUE VARCHAR2,INCLUDE_NON_NAV BOOLEAN DEFAULT TRUE) IS 
----------------------------------------------------------------------------------------------
		ITM_CANVAS	   VARCHAR2(80);
		CURR_ITM       VARCHAR2(80);
		ENANLED_ITEM       VARCHAR2(80);
		RESTRICT_PROP  NUMBER;
BEGIN
		CURR_ITM    := Get_Block_Property(cur_blk , FIRST_ITEM);
		IF CURR_ITM IS NULL THEN
				RETURN;
		ELSE
	  	  ITM_CANVAS := Get_Item_Property(cur_blk || '.' || CURR_ITM,ITEM_CANVAS);
	   		IF INCLUDE_NON_NAV THEN
	   				ENANLED_ITEM   := 'TRUE';

	   		END IF;

							Set_Item_Property(cur_blk || '.' || CURR_ITM,PROP,PROP_VALUE);
	   		END IF;	
		END IF;

		WHILE CURR_ITM IS NOT NULL LOOP
				CURR_ITM    := Get_Item_Property(cur_blk || '.' || CURR_ITM,NEXTITEM);
				IF CURR_ITM IS NOT NULL THEN 
		   			ITM_CANVAS := Get_Item_Property(cur_blk || '.' || CURR_ITM,ITEM_CANVAS);
	   				IF INCLUDE_NON_NAV THEN
	   						ENANLED_ITEM   := 'TRUE';
	   				ELSE
	   						ENANLED_ITEM   := Get_Item_Property(cur_blk || '.' || CURR_ITM,NAVIGABLE);
	   				END IF;
		   			IF ITM_CANVAS IS NOT NULL AND ENANLED_ITEM = 'TRUE'  THEN
								Set_Item_Property(cur_blk || '.' || CURR_ITM,PROP,PROP_VALUE);
		   			END IF;
				END IF;
		END LOOP;
END SET_BLOCK_ITEMS;

--	SHOW OR HIDE THE TAB PAGE AND MAKE ALL ITEMS
--	IN THE PAGE NONE NAVIGABLE 
----------------------------------------------------------------------------------------------
PROCEDURE SHOW_TAB_PAGE(curform VARCHAR2,P_TAB_PAGE VARCHAR2,SHOW_FLAG BOOLEAN) IS 
----------------------------------------------------------------------------------------------
		ITM_CANVAS	  VARCHAR2(80);
		CURR_ITM      VARCHAR2(80);
		blkname				VARCHAR2(80);
		VISIBLE_VAL		VARCHAR2(10);
		ENANLED_VAL   VARCHAR2(10);
		RESTRICT_PROP NUMBER;
		PROP_VAL			NUMBER;
		V_ITEM_TYPE VARCHAR2(200);
BEGIN
		IF SHOW_FLAG THEN
			PROP_VAL := PROPERTY_TRUE;
		ELSE
			PROP_VAL := PROPERTY_FALSE;
		END IF;

		blkname := Get_Form_Property(curform,FIRST_BLOCK);
		WHILE blkname IS NOT NULL LOOP
				CURR_ITM    := Get_Block_Property(blkname , FIRST_ITEM);
				WHILE CURR_ITM IS NOT NULL LOOP
			  	  V_ITEM_TYPE := Get_Item_Property(blkname || '.' || CURR_ITM, ITEM_TYPE);

			  	  ITM_CANVAS  := Get_Item_Property(blkname || '.' || CURR_ITM,ITEM_TAB_PAGE);
			  	  IF V_ITEM_TYPE IN ('CHECKBOX','LIST', 'RADIO', 'TEXT ITEM') THEN
				  	  ENANLED_VAL := Get_Item_Property(blkname || '.' || CURR_ITM,ENABLED);
			  	  ELSE
			  	  	ENANLED_VAL := 'FALSE';
			  	  END IF;

	  			  VISIBLE_VAL := Get_Item_Property(blkname || '.' || CURR_ITM,VISIBLE);
 	  				IF UPPER(ITM_CANVAS) = UPPER(P_TAB_PAGE) AND ENANLED_VAL = 'TRUE' AND VISIBLE_VAL = 'TRUE' THEN
 	  						Set_Item_Property(blkname || '.' || CURR_ITM,NAVIGABLE,PROP_VAL);
 			  		END IF;							
						CURR_ITM    := Get_Item_Property(blkname || '.' || CURR_ITM,NEXTITEM);
				END LOOP;			  	  
				blkname := Get_BLOCK_Property(blkname,NEXTBLOCK);				
		END LOOP;
		SET_TAB_PAGE_PROPERTY(P_TAB_PAGE,VISIBLE,PROP_VAL);
END SHOW_TAB_PAGE;

	FUNCTION COMP_OPRT(V_VAL1 VARCHAR2, V_COMP_OPRTN VARCHAR2, V_ORG_VAL2 IN VARCHAR2) RETURN BOOLEAN IS
--------------------------------------------------------------------------------    
	V_VAL2 VARCHAR2(2000) := V_ORG_VAL2;
	
	BEGIN
		IF 		V_COMP_OPRTN LIKE '=' THEN
			IF V_VAL1 = V_VAL2 THEN
				RETURN TRUE;
			ELSE
				RETURN FALSE;
			END IF;
		ELSIF V_COMP_OPRTN LIKE '<>' THEN 
			IF V_VAL1 <> V_VAL2 THEN
				RETURN TRUE;
			ELSE
				RETURN FALSE;
			END IF;
		ELSIF V_COMP_OPRTN LIKE '>' THEN 
			IF V_VAL1 > V_VAL2 THEN
				RETURN TRUE;
			ELSE
				RETURN FALSE;
			END IF;
		ELSIF V_COMP_OPRTN LIKE '<' THEN 
			IF V_VAL1 < V_VAL2 THEN
				RETURN TRUE;
			ELSE
				RETURN FALSE;
			END IF;
		ELSIF V_COMP_OPRTN LIKE 'IS NULL' THEN 
			IF V_VAL1 IS NULL THEN
				RETURN TRUE;
			ELSE
				RETURN FALSE;
			END IF;
		ELSIF V_COMP_OPRTN LIKE 'IS NOT NULL' THEN 
			IF V_VAL1 IS NOT NULL THEN
				RETURN TRUE;
			ELSE
				RETURN FALSE;
			END IF;
			ELSIF V_COMP_OPRTN LIKE 'IN' THEN 
				IF IN_LIST(V_VAL1, V_VAL2) THEN
					RETURN TRUE;
				ELSE
					RETURN FALSE;
				END IF;
		ELSIF V_COMP_OPRTN LIKE 'NOT IN' THEN 
				IF IN_LIST(V_VAL1, V_VAL2) THEN
					RETURN FALSE;
				ELSE
					RETURN TRUE;
				END IF;
		ELSE
			STR.DISPALERT('Œÿ√ ›Ì ‰Ê⁄ ⁄„·ÌÂ «·„ﬁ«—‰Â','ERROR');
			RAISE FORM_TRIGGER_FAILURE;
		END IF;
	END COMP_OPRT;

--------------------------------------------------------------------------------    
	FUNCTION VALID_COMP_OPRT(V_OPRT VARCHAR2) RETURN BOOLEAN IS

			RETURN TRUE;
		ELSE
			RETURN FALSE;
		END IF;
		
	END VALID_COMP_OPRT;

--------------------------------------------------------------------------------    
	FUNCTION IN_LIST(V_VAL1 VARCHAR2, V_ORG_VAL2 IN VARCHAR2) RETURN BOOLEAN IS
--------------------------------------------------------------------------------    
		LIST_COUNT NUMBER;
		FOUND BOOLEAN := FALSE;
		LIST_VAL VARCHAR2(2000);
		COUNTER NUMBER := 1;
		V_VAL2 VARCHAR2(2000) := V_ORG_VAL2;
	BEGIN
		V_VAL2 := REPLACE(V_VAL2, '(');
		V_VAL2 := REPLACE(V_VAL2, ')');
		V_VAL2 := ','||V_VAL2||',';	

		LIST_COUNT := LENGTH(V_VAL2) - LENGTH(REPLACE(V_VAL2,',')) - 1;
	
		WHILE COUNTER <= LIST_COUNT AND NOT FOUND LOOP
			LIST_VAL := SUBSTR(V_VAL2, INSTR(V_VAL2, ',', 1, COUNTER)+1, INSTR(V_VAL2, ',',1, COUNTER+1)- INSTR(V_VAL2, ',', 1, COUNTER) -1);
			LIST_VAL := LTRIM(LIST_VAL);
			LIST_VAL := RTRIM(LIST_VAL);
			
			IF V_VAL1 = LIST_VAL THEN
				FOUND := TRUE;
			END IF;
			COUNTER := COUNTER + 1;		
		END LOOP;
		
		RETURN FOUND;
	END IN_LIST;

END LNK ;

   FUNCTION CHECK_EDI_ROLE_ID (V_PORT_CD NUMBER, V_USER_ID VARCHAR2)
      RETURN BOOLEAN
   IS
   V_ROLE_ID   VARCHAR2 (30);
   V_RETURN    BOOLEAN;
BEGIN
   BEGIN
     SELECT UR.ROLE_ID INTO V_ROLE_ID
     FROM SC_USER_ROLES UR
     WHERE  UR.USER_ID  = V_USER_ID
     AND  UR.PORT_CD   = V_PORT_CD
     AND   UR.UR_ACTIVE  = 1
     AND  UR.ROLE_ID   = 'EDIMSG_ALL'
    UNION
     SELECT  GR.ROLE_ID
     FROM   SC_GROUP_ROLES GR, SC_USER_GROUPS UG
     WHERE   UG.PORT_CD   = V_PORT_CD
     AND   UG.USER_ID   = V_USER_ID
     AND   GR.ROLE_ID   = 'EDIMSG_ALL'
     AND   GR.GROUP_ID   = UG.GROUP_ID
     AND   GR.GR_ACTIVE  = 1
    MINUS
     SELECT ROLE_ID  FROM SC_USER_EXCLUDED_ROLES UER
     WHERE  UER.USER_ID = V_USER_ID
     AND   UER.PORT_CD = V_PORT_CD
     AND   UER.ROLE_ID = 'EDIMSG_ALL'
     AND   UER.UER_ACTIVE =1
     ;

      EXCEPTION WHEN NO_DATA_FOUND THEN RETURN(FALSE);
   END;
     IF V_ROLE_ID = 'EDIMSG_ALL' THEN
      RETURN(TRUE);
     ELSE
      RETURN(FALSE);
     END IF; 

   RETURN V_RETURN;
END;

           Delete_Timer('RETURN');

              To_Number(Get_View_Property('EDI_MSG', DISPLAY_X_POS))+0.006);
        end if;
     COPY('2', 'GLOBAL.TIMER');

FUNCTION CHECK_IS_ITEM_LOW_PRICE (PORT_CD2 NUMBER,DCLTN_ISN2 NUMBER,ITEM_ISN2 NUMBER,ShowErrMsg boolean DEFAULT TRUE)
RETURN INTEGER IS
	dcltn_item_rec 	DCLTN_ITEMS%ROWTYPE ;
	HRMNZD_CODE2 	VARCHAR2(8);
	I 					  INTEGER:=0;
	HRMNZ_LEN	   	INTEGER:=8;
	MIN_CIF_COST	NUMBER(14,3):=0;
	MIN_CIF_COST2	NUMBER(14,3):=0;
  HRMNZD_TYPE2	NUMBER(1);
  CNT						NUMBER;	
BEGIN
	SELECT * INTO dcltn_item_rec
	FROM DCLTN_ITEMS
	WHERE PORT_CD = PORT_CD2
	AND ITEM_ISN = ITEM_ISN2;
	HRMNZD_CODE2 := IS_PRICING_TARIFF_CD(dcltn_item_rec.FINAL_TARIFF_CD);
	IF HRMNZD_CODE2 IS NOT NULL THEN
		--TO GET HRMNZD_TYPE
		SELECT HRMNZD_TYPE INTO HRMNZD_TYPE2
		FROM ITEM_VALUE_CHECKS
		WHERE HRMNZD_CODE = HRMNZD_CODE2;
		-- TO CHECK HRMZD TYPE 
		IF HRMNZD_TYPE2 = 1 OR HRMNZD_TYPE2=2 THEN
			HRMNZ_LEN := HRMNZD_TYPE2 * 2;
		END IF;
		--dcltn_item_rec.CIF_COST/dcltn_item_rec.INTL_QTY
		SELECT COUNT(VIV.PORT_CD) INTO CNT
		FROM VALUE_ITEM_VALUES VIV,VALUE_ITEM_INFOS VII
		WHERE VIV.PORT_CD = VII.PORT_CD
		AND VIV.ITEM_ISN = VII.ITEM_ISN
		AND VII.HRMNZD_CODE =  HRMNZD_CODE2;
		
		IF CNT >0 THEN
				SELECT MIN(VIV.CIF_COST) INTO MIN_CIF_COST
				FROM VALUE_ITEM_VALUES VIV,VALUE_ITEM_INFOS VII
				WHERE VIV.PORT_CD = VII.PORT_CD
				AND VIV.ITEM_ISN = VII.ITEM_ISN
				AND VII.HRMNZD_CODE=  HRMNZD_CODE2;
		END IF;
		MIN_CIF_COST2 :=dcltn_item_rec.CIF_COST/dcltn_item_rec.INTL_QTY;
		--STR.DISPALERT('MIN_CIF_COST=' ||MIN_CIF_COST ||' DCLTN CIFCOST=' || MIN_CIF_COST2,'ERROR' );
		IF MIN_CIF_COST > MIN_CIF_COST2 AND MIN_CIF_COST > 0 THEN
			I:=1;
			If (ShowErrMsg) Then
				STR.DispArbMsg(22,40,'ERROR');
			End if;
		Else
				If (ShowErrMsg) Then
					STR.DispArbMsg(22,100,'INFORMATION');
				End if;
		END IF;
	ELSE
		--STR.DSIPALERT('»‰œ Â–« «·’‰› €Ì— „œ—Ã „‰ ÷„‰ »‰Êœ «·ﬁÌ„…','INFORMATION');--
		If (ShowErrMsg) THEN
			STR.DispArbMsg(22,60,'ERROR');
		End If;
	END IF;
	RETURN I;
END;

/********************************************************************************************/
/*		IS_PRICING_TARIFF_CD																																	*/
/********************************************************************************************/
FUNCTION IS_PRICING_TARIFF_CD (TARIFF_CD VARCHAR2)
RETURN VARCHAR2 IS
	HRMNZD_CD VARCHAR2(8):=NULL;
BEGIN

	BEGIN
	SELECT HRMNZD_CODE INTO HRMNZD_CD
	FROM ITEM_VALUE_CHECKS
	WHERE HRMNZD_CODE = TARIFF_CD;
	IF HRMNZD_CD IS NOT NULL THEN
		RETURN HRMNZD_CD;
	END IF;
	EXCEPTION WHEN OTHERS THEN HRMNZD_CD :=NULL; 
	END;

	
	BEGIN
	SELECT HRMNZD_CODE INTO HRMNZD_CD
	FROM ITEM_VALUE_CHECKS
	WHERE SUBSTR(HRMNZD_CODE,1,4) = SUBSTR(TARIFF_CD,1,4)
	AND HRMNZD_TYPE = 2;
	IF HRMNZD_CD IS NOT NULL THEN
		RETURN HRMNZD_CD;
	END IF;
	EXCEPTION WHEN OTHERS THEN HRMNZD_CD :=NULL; 
	END;
	BEGIN
	SELECT HRMNZD_CODE INTO HRMNZD_CD
	FROM ITEM_VALUE_CHECKS
	WHERE SUBSTR(HRMNZD_CODE,1,2) = SUBSTR(TARIFF_CD,1,2)
	AND HRMNZD_TYPE = 1;
	IF HRMNZD_CD IS NOT NULL THEN
		RETURN HRMNZD_CD;
	END IF;
	EXCEPTION WHEN OTHERS THEN HRMNZD_CD :=NULL; 
	END;
	RETURN HRMNZD_CD;
END;

/********************************************************************************************/
/*	GET_DCLTN_OTHER_EXPNS																																		*/
/********************************************************************************************/
FUNCTION GET_DCLTN_OTHER_EXPNS(PORT_CD2 NUMBER,DCLTN_ISN2 NUMBER)
RETURN NUMBER IS
	CURSOR crsr_Dcltninv IS
	SELECT *
	FROM  DCLTN_INVOICES
	WHERE PORT_CD = PORT_CD2
	AND DCLTN_ISN = DCLTN_ISN2;

	OTHER_COST NUMBER(15,3);
	SHPMT_EXPS	NUMBER;
	DCLTN_DT DATE;
BEGIN
	-- to get dcltn date
	BEGIN
	SELECT DCLTN_DT INTO DCLTN_DT
	FROM DECLARATIONS
	WHERE PORT_CD = PORT_CD2
	AND DCLTN_ISN = DCLTN_ISN2;
	EXCEPTION WHEN OTHERS THEN STR.DispAlert(DBMS_ERROR_TEXT || '!!  GET_DCLTN_OTHER_EXPNS FUNC DCLTN_DT','ERROR');
	END;
	OTHER_COST :=0;
	FOR dcltn_inv_rec IN crsr_Dcltninv LOOP
		IF (NVL(dcltn_inv_rec.SHPMT_EXPS,0)> NVL(dcltn_inv_rec.AIR_SHPMT_EXPS,0)) THEN
			SHPMT_EXPS :=dcltn_inv_rec.SHPMT_EXPS;
		ELSE
			SHPMT_EXPS :=NVL(dcltn_inv_rec.AIR_SHPMT_EXPS,0);
		END IF;
		OTHER_COST := OTHER_COST + SHPMT_EXPS *	REVENUE.GET_CURRENCY_RATE(dcltn_inv_rec.SHPMT_CURCY_CD,DCLTN_DT);
		OTHER_COST := OTHER_COST + dcltn_inv_rec.INS_EXPS *	REVENUE.GET_CURRENCY_RATE(dcltn_inv_rec.INS_CURCY_CD,DCLTN_DT);
		OTHER_COST := OTHER_COST + dcltn_inv_rec.OTHER_EXPS *	REVENUE.GET_CURRENCY_RATE(dcltn_inv_rec.OTHER_EXPS,DCLTN_DT);
	END LOOP;
	RETURN OTHER_COST;
	EXCEPTION WHEN OTHERS THEN STR.DispAlert(DBMS_ERROR_TEXT || '!!  GET_DCLTN_OTHER_EXPNS FUNC ','ERROR');
END;

/********************************************************************************************/
/*	IS_DCLTN_EXISTS_IN_PRICING																															*/
/********************************************************************************************/	
FUNCTION IS_DCLTN_EXISTS_IN_PRICING(V_PORT_CD NUMBER,V_ITEM_ISN NUMBER:=NULL
					 ,V_INV_ISN  NUMBER  :=NULL,V_DCLTN_ISN NUMBER  :=NULL )	RETURN INTEGER IS
	V_COUNT NUMBER:=0;
BEGIN
	IF  V_ITEM_ISN IS NOT NULL THEN
		SELECT COUNT(ITEM_ISN) 	INTO V_COUNT
		FROM VALUE_DCLTN_PRMTRS
		WHERE PORT_CD = V_PORT_CD
		AND ITEM_ISN = V_ITEM_ISN ;
	ELSIF V_INV_ISN  IS NOT NULL THEN
		SELECT COUNT(ITEM_ISN) 	INTO V_COUNT
		FROM VALUE_DCLTN_PRMTRS
		WHERE PORT_CD = V_PORT_CD
		AND ITEM_ISN IN
		( SELECT ITEM_ISN
		  FROM DCLTN_ITEMS
		  WHERE INV_ISN = V_INV_ISN);
	ELSIF V_DCLTN_ISN IS NOT NULL THEN
		SELECT COUNT(ITEM_ISN) 	INTO V_COUNT
		FROM VALUE_DCLTN_PRMTRS
		WHERE PORT_CD = V_PORT_CD
		AND DCLTN_ISN = V_DCLTN_ISN ;
	END IF;
	IF V_COUNT=0 THEN
		RETURN 0;
	ELSE
		RETURN 1;
	END IF;
END;

FUNCTION  IF_DCLTN_CAN_ADD_TO_PRC (V_PORT_CD NUMBER,V_DCLTN_ISN NUMBER)
RETURN INTEGER IS
	CURSOR crsr_DcltnItem IS 
	SELECT * 
	FROM DCLTN_ITEMS
	WHERE PORT_CD = V_PORT_CD
	AND 	DCLTN_ISN = V_DCLTN_ISN;
	HRMNZD_CODE2 	VARCHAR2(8);
BEGIN
	FOR item_DTLS_rec IN crsr_DcltnItem LOOP
		HRMNZD_CODE2 := IS_PRICING_TARIFF_CD(item_DTLS_rec.FINAL_TARIFF_CD);
		IF HRMNZD_CODE2 IS NOT NULL THEN
				RETURN 1;	
		END IF ;
	END LOOP;
	RETURN 0;	
END;

	SELECT FINAL_TARIFF_CD INTO V_FINAL_TARIFF_CD

    PROCEDURE Handle_run_product(product NUMBER,reportFile VARCHAR2,commMode NUMBER,engine NUMBER,fileLocation NUMBER,mparamlist_id ParamList) IS

	  rp2rroExecution_mode         number			   := Runtime;
   	rp2rro_jobidFull	           varchar2(100);
	  rp2rro_status                varchar2(100);
	  rp2rro_jobidPartial 				 varchar2(100);
	  v_destype_type               varchar2(100); 
	  v_destype_value              varchar2(100); 
	   v_report_name VARCHAR2(4000);
  BEGIN
  	
  	IF INSTR(reportFile, '/') > 0 OR INSTR(reportFile, '\') > 0 THEN
       v_report_name := SUBSTR(
           reportFile,
           GREATEST(
               INSTR(reportFile, '/', -1),
               INSTR(reportFile, '\', -1)
           ) + 1
       );
   ELSE
       v_report_name := reportFile;
   END IF;

    	--	SET_REPORT_OBJECT node properties
 	 	  SET_REPORT_OBJECT_PROPERTY(rp2rroNodeId,REPORT_FILENAME,v_report_name);--reportFile
 	 		--SET_REPORT_OBJECT_PROPERTY(rp2rroNodeId,REPORT_DESNAME,rp2rroDesname);
 	 	  SET_REPORT_OBJECT_PROPERTY(rp2rroNodeId,REPORT_EXECUTION_MODE,rp2rroExecution_mode);
  	  SET_REPORT_OBJECT_PROPERTY(rp2rroNodeId,REPORT_COMM_MODE,rp2rroComm_mode);
		
			-- setting the Reports services output format. If you want to have the user to decide about the Reports output
			-- format then you need set "rp2rroDestype" variable to a value determined by a user selectable item
  	  IF (upper(rp2rroDestype)='CACHE') THEN
  	  	SET_REPORT_OBJECT_PROPERTY(rp2rroNodeId,REPORT_DESTYPE,CACHE);
  	  
  	  ELSIF (upper(rp2rroDestype)='PRINTER') THEN
  	  	--SET_REPORT_OBJECT_PROPERTY(rp2rroNodeId,REPORT_DESTYPE,PRINTER);
  	  	SET_REPORT_OBJECT_PROPERTY(rp2rroNodeId,REPORT_DESTYPE,CACHE);
  	  ELSIF (upper(rp2rroDestype)='FILE') THEN
  	  	SET_REPORT_OBJECT_PROPERTY(rp2rroNodeId,REPORT_DESTYPE,FILE);
  	  ELSIF (upper(rp2rroDestype)='MAIL') THEN
  	    SET_REPORT_OBJECT_PROPERTY(rp2rroNodeId,REPORT_DESTYPE,MAIL);
  	  ELSE
  	  	-- default
  	  	message('Not a valid destype');
  	  	null;
  	  END IF;
 	  		
  	  SET_REPORT_OBJECT_PROPERTY(rp2rroNodeId,REPORT_DESFORMAT,rp2rroDesformat);
  	  SET_REPORT_OBJECT_PROPERTY(rp2rroNodeId,REPORT_SERVER,rp2rroReportServer);
 	  
  	  IF id_null (mparamlist_id)  THEN
  	  	
  	  	SET_REPORT_OBJECT_PROPERTY(rp2rroNodeId,REPORT_OTHER,rp2rroReportOther);
		   	rp2rro_jobidFull:=run_report_object(rp2rroNodeId);
  	  ELSE
  	  	GET_PARAMETER_ATTR(mparamlist_id, 'DESTYPE',v_destype_type,v_destype_value);
  	  	IF UPPER(v_destype_value) ='SCREEN' THEN
  	  			SET_REPORT_OBJECT_PROPERTY(rp2rroNodeId,REPORT_DESTYPE,CACHE);
  	  	ELSIF UPPER(v_destype_value) ='PRINTER' THEN
  	  		  SET_REPORT_OBJECT_PROPERTY(rp2rroNodeId,REPORT_DESTYPE,CACHE);
  	  			--SET_REPORT_OBJECT_PROPERTY(rp2rroNodeId,REPORT_DESTYPE,PRINTER);
  	  	ELSE
  	  		SET_REPORT_OBJECT_PROPERTY(rp2rroNodeId,REPORT_DESTYPE,CACHE);
  	  	END IF;
  	  	
  	    rp2rro_jobidFull:=run_report_object(rp2rroNodeId,mparamlist_id);
  	  END IF;	  
  	  -- If communication mode Synchronous then provide Reports success or error status
  	  
  	  IF (rp2rroComm_mode = SYNCHRONOUS) THEN  	 
  	  	-- Check the Reports Status.
 
  			-- the partial job id is used to retrieve the Reports output. It contain sthe Reportsjob_id only

   	  	rp2rro_jobidPartial := substr(rp2rro_jobidFull,length(rp2rroReportServer)+2,length(rp2rro_jobidFull));
      	rp2rro_status:= report_object_status(rp2rro_jobidFull);

  	  		IF (rp2rro_status='FINISHED') then		 
		  			/* Report run was successful 
		     		Get the Report output to the Browser */
		    			IF (upper(rp2rroDestype) ='CACHE') THEN
		     			 web.show_document('http://10.55.196.159:9002'||rp2rroVirtualDir||rp2rroReportsInterface||'/getjobid'||rp2rro_jobidPartial||'?server='||rp2rroReportServer,'_blank');         
		    			ELSE
		      		-- Reports output is send to printer or file, there is nothing to download to the client
		      		null; 
		    			END IF;
		  		ELSE -- Report generation errors
	     	   				message('Report failed with message: '||rp2rro_status);
		  		END IF;
		  END IF;
   END;

	CURSOR 	FIRST_STMT1_FLDS(V_STMT1_SEQ NUMBER, V_FLD_TYPE NUMBER) IS
	  			SELECT *
	  			FROM 	CON_FIELDS
					WHERE	FIELD_TYPE	= V_FLD_TYPE
					AND 	CON1_SEQ 		= V_STMT1_SEQ
					AND 	REC_STATUS 	= 1					
					AND		CON2_SEQ 	IS NULL
					AND		LOGICAL_OPRTN IS NULL					
					;

--------------------------------------------------------------------------------    
 	FUNCTION	GET_STMT_RSLT(V_STMT1_SEQ NUMBER) RETURN BOOLEAN IS 
--------------------------------------------------------------------------------    
		PREV_RESULT BOOLEAN;		
		CURR_RESULT BOOLEAN;		
		FIRST_FLDS_REC CON_FIELDS%ROWTYPE;
		V_COUNT NUMBER := 0;
	BEGIN

	OPEN FIRST_STMT1_FLDS(V_STMT1_SEQ, 1);

			CURR_RESULT := GET_SCRN_RSLT(FIRST_FLDS_REC);
			PREV_RESULT := LNK.LOGIC_OPRT(PREV_RESULT, 'AND' ,CURR_RESULT); 
		END IF;

		FETCH FIRST_STMT1_FLDS INTO FIRST_FLDS_REC;
	END LOOP;
	CLOSE FIRST_STMT1_FLDS;
	
	IF V_COUNT = 0 THEN
		STR.DISPALERT(' ⁄—Ì› ‘—Êÿ ⁄—÷ «·ÊŸ«∆› €Ì— ’ÕÌÕ - ·«  ÊÃœ ÕﬁÊ· ‘«‘«  „ÕœœÂ','ERROR');
		RAISE FORM_TRIGGER_FAILURE;
	END IF;

	RETURN PREV_RESULT;

	FUNCTION	GET_SCRN_RSLT(V_FIRST_FLDS_REC CON_FIELDS%ROWTYPE)RETURN BOOLEAN IS
--------------------------------------------------------------------------------    
	SCRN_NAME VARCHAR2(200);
	SCRN_VAL	 VARCHAR2(200);
	COMP_VAL	 VARCHAR2(200);		
	RES BOOLEAN;
	BEGIN
		IF V_FIRST_FLDS_REC.DATA_SRCR_NAME IS NULL  OR V_FIRST_FLDS_REC.FLD_NAME IS NULL  THEN

		END IF;
		
		SCRN_NAME := ''||V_FIRST_FLDS_REC.DATA_SRCR_NAME||'.'||V_FIRST_FLDS_REC.FLD_NAME||'';
		SCRN_VAL := NAME_IN(SCRN_NAME);

		IF V_FIRST_FLDS_REC.COMP_VALUE IS NULL THEN
			IF 	V_FIRST_FLDS_REC.COMP_OPERATOR NOT IN ('IS NULL', 'IS NOT NULL') THEN

		ELSE

						STR.DISPALERT(' ⁄—Ì› ‘—Êÿ ⁄—÷ «·ÊŸ«∆› €Ì— ’ÕÌÕ - ﬁÌ„… «·„ﬁ«—‰Â ·Õﬁ· «·‘«‘Â €Ì— ’ÕÌÕÂ','ERROR');
						RAISE FORM_TRIGGER_FAILURE;
			END IF;
		END IF;

		IF V_FIRST_FLDS_REC.COMP_VALUE_TYPE IN (1, 5) THEN -- CONSTANT 1 OR LIST 5
			COMP_VAL := V_FIRST_FLDS_REC.COMP_VALUE;
		ELSIF V_FIRST_FLDS_REC.COMP_VALUE_TYPE = 2 THEN -- SCREEN BLOCK
			COMP_VAL := V_FIRST_FLDS_REC.COMP_VALUE;
			COMP_VAL := NAME_IN(COMP_VAL);
		ELSIF V_FIRST_FLDS_REC.COMP_VALUE_TYPE = 3 THEN -- PARAMETER
			COMP_VAL := V_FIRST_FLDS_REC.COMP_VALUE;
			COMP_VAL := NAME_IN(COMP_VAL);
		ELSE

				RAISE FORM_TRIGGER_FAILURE;
			END IF;
		END IF;

		RES := LNK.COMP_OPRT(SCRN_VAL, V_FIRST_FLDS_REC.COMP_OPERATOR , COMP_VAL);
		
		RETURN RES;
	END GET_SCRN_RSLT;

	FUNCTION DISP_STATS_FUNS(V_STATS_TYPE NUMBER, V_STATS_STATUS NUMBER, V_TASK_ID NUMBER, V_STATS_LEVEL NUMBER) RETURN BOOLEAN;

PACKAGE BODY STATS_FN IS

	CURSOR STATS_FN_CD(V_TYPE NUMBER, V_STATUS NUMBER, V_TASK NUMBER, V_LEVEL NUMBER) IS
	  			SELECT DISTINCT FUN_CODE 
	  			FROM 	STATS_CTRLS
	  			WHERE (TASK_ID IS NULL OR TASK_ID = V_TASK)
	  			AND		(LEVEL_ID IS NULL OR LEVEL_ID = V_LEVEL)
	  			AND		(STATS_TYPE IS NULL OR STATS_TYPE = V_TYPE)
	  			AND		(STATS_STA IS NULL OR STATS_STA = V_STATUS)
	  			AND		REC_STATUS = 1
	  			;
---------------------------------------------------------------------------------------	
	PROCEDURE ADJ_STATS_FUNS(V_STATS_TYPE NUMBER, V_STATS_STATUS NUMBER, V_TASK_ID NUMBER, V_STATS_LEVEL NUMBER) IS
---------------------------------------------------------------------------------------	
	BEGIN
		TMPLT.DISABLE_ALL_FUNS;
		
		FOR I IN STATS_FN_CD(V_STATS_TYPE , V_STATS_STATUS , V_TASK_ID , V_STATS_LEVEL) LOOP
	 		TMPLT.ENABLE_FUN(I.FUN_CODE);
		END LOOP;			

  	TMPLT.ENB_DIS_FUN_BUT ;

  END ADJ_STATS_FUNS;
  
---------------------------------------------------------------------------------------	
	FUNCTION DISP_STATS_FUNS(V_STATS_TYPE NUMBER, V_STATS_STATUS NUMBER, V_TASK_ID NUMBER, V_STATS_LEVEL NUMBER) RETURN BOOLEAN IS
---------------------------------------------------------------------------------------	
	STATS_FN_CD_REC STATS_FN_CD%ROWTYPE;
	BEGIN
		OPEN STATS_FN_CD(V_STATS_TYPE , V_STATS_STATUS , V_TASK_ID , V_STATS_LEVEL);
		FETCH STATS_FN_CD INTO STATS_FN_CD_REC;
		
		IF STATS_FN_CD%FOUND THEN
			CLOSE STATS_FN_CD;
			RETURN TRUE;
		ELSE
			CLOSE STATS_FN_CD;
			RETURN FALSE;
		END IF;
  END DISP_STATS_FUNS;

END STATS_FN;

    WHERE var_name = 'PORT_CD';

   v_audit_sys := risk_audit.get_system (v_obj_id);
   v_audit_isn :=
                risk_audit.audit_msg (v_obj_id, p_ramsg_msg, p_ramsg_msg_type);

   SELECT racnfg_isn
     INTO v_racnfg_isn
     FROM rsk_audit_msg
    WHERE port_cd = v_port_cd AND audit_isn = v_audit_isn;

   FOR curr IN (SELECT raf.racnfg_field
                  FROM rsk_audit_field raf
                 WHERE raf.racnfg_fld_status = 1
                   AND raf.racnfg_isn = v_racnfg_isn)
   LOOP
      risk_audit.insert_audit_dlts (v_port_cd,
                                    v_audit_sys,
                                    v_obj_id,
                                    v_audit_isn,
                                    curr.racnfg_field,
                                     name_in(curr.racnfg_field)
                                   );
      
   END LOOP;
EXCEPTION
   WHEN OTHERS
   THEN
      NULL;
END rsk_audit;

--------------------------------------------------------------------------------    
	FUNCTION GET_QRY_RSLT RETURN NUMBER IS
--------------------------------------------------------------------------------    
	SQL_STMT VARCHAR2(2000);

	ROW_COUNT	NUMBER := 0;

	CURR_RES 		BOOLEAN;
	PREV_RES 		BOOLEAN;	
	ROW_RESULT	BOOLEAN;
	RET_VAL			NUMBER := -1;

	CURSOR_ID INTEGER;	
	ERROR			INTEGER;			

	BEGIN

	SQL_STMT := 'SELECT '||SELECT_FIELDS||' FROM '||TABLES_NAMES||' WHERE '||WHERE_FIELDS;
	CURSOR_ID := DBMS_SQL.OPEN_CURSOR;
	DBMS_SQL.PARSE(CURSOR_ID, SQL_STMT, 1);
	FOR I IN 1..TBL_FLD_CNT LOOP
		DBMS_SQL.DEFINE_COLUMN(CURSOR_ID, I, FLD_COMP_VALUES(I).FLD_NAME, 200);
	END LOOP;

	ERROR := DBMS_SQL.EXECUTE(CURSOR_ID);

	WHILE (DBMS_SQL.FETCH_ROWS(CURSOR_ID) > 0 ) LOOP
		ROW_COUNT := ROW_COUNT + 1;

		FOR I IN 1..TBL_FLD_CNT LOOP
			DBMS_SQL.COLUMN_VALUE(CURSOR_ID, I, FLD_COMP_VALUES(I).FLD_VAL);

			IF I = 1 THEN
				PREV_RES := LNK.COMP_OPRT(FLD_COMP_VALUES(I).FLD_VAL, FLD_COMP_VALUES(I).COMP_OPRT, FLD_COMP_VALUES(I).FLD_COMP_VAL);
			ELSE
				CURR_RES := LNK.COMP_OPRT(FLD_COMP_VALUES(I).FLD_VAL, FLD_COMP_VALUES(I).COMP_OPRT, FLD_COMP_VALUES(I).FLD_COMP_VAL);			
				PREV_RES := LNK.LOGIC_OPRT(PREV_RES, 'AND', CURR_RES);
			END IF;

		END LOOP;
		
		IF ROW_COUNT = 1 THEN
			ROW_RESULT := PREV_RES;
		ELSIF ROW_COUNT > 1 THEN
			ROW_RESULT := LNK.LOGIC_OPRT(ROW_RESULT, 'AND', PREV_RES);
		END IF;

	END LOOP;
	DBMS_SQL.CLOSE_CURSOR(CURSOR_ID);
	
	IF ROW_RESULT = TRUE THEN
		RET_VAL := 1;
	ELSIF ROW_RESULT = FALSE THEN
		RET_VAL := 0;
	ELSE
		RET_VAL := -1;
	END IF;

	RETURN RET_VAL;

	 EXCEPTION
  	 WHEN OTHERS  THEN 
  	     IF (DBMS_SQL.IS_OPEN(CURSOR_ID)) THEN 
  	     	   DBMS_SQL.CLOSE_CURSOR(CURSOR_ID);
  	     END IF;
				 STR.DISPALERT('Œÿ√ ›Ì Ã„·… ≈Œ »«— «·‘—Êÿ ›Ì ﬁ«⁄œ… «·»Ì«‰«  ⁄‰œ  ‰›Ì–Â«','ERROR');
  	     RETURN 0;

	END GET_QRY_RSLT ;

	PROCEDURE FILL_TBL_VALUES(V_TBL_FLD_CNT NUMBER, V_FLDS_REC CON_FIELDS%ROWTYPE) IS
--------------------------------------------------------------------------------    
	BEGIN
		IF V_FLDS_REC.COMP_VALUE IS NULL THEN
			IF 	V_FLDS_REC.COMP_OPERATOR NOT IN ('IS NULL', 'IS NOT NULL') THEN			
						STR.DISPALERT(' ⁄—Ì› ‘—Êÿ ⁄—÷ «·ÊŸ«∆› €Ì— ’ÕÌÕ - ﬁÌ„… «·„ﬁ«—‰Â ·Õﬁ· «·ÃœÊ· €Ì— ’ÕÌÕÂ','ERROR');
						RAISE FORM_TRIGGER_FAILURE;
			END IF;
		ELSE
			IF 	V_FLDS_REC.COMP_OPERATOR IN ('IS NULL', 'IS NOT NULL') THEN			
						STR.DISPALERT(' ⁄—Ì› ‘—Êÿ ⁄—÷ «·ÊŸ«∆› €Ì— ’ÕÌÕ - ﬁÌ„… «·„ﬁ«—‰Â ·Õﬁ· «·ÃœÊ· €Ì— ’ÕÌÕÂ','ERROR');
						RAISE FORM_TRIGGER_FAILURE;
			END IF;
		END IF;

		FLD_COMP_VALUES(V_TBL_FLD_CNT).FLD_NAME		:=	V_FLDS_REC.DATA_SRCR_NAME||'.'||V_FLDS_REC.FLD_NAME;
		FLD_COMP_VALUES(V_TBL_FLD_CNT).COMP_OPRT	:=	V_FLDS_REC.COMP_OPERATOR;
		
		IF V_FLDS_REC.COMP_VALUE_TYPE IN (1, 5) THEN -- CONSTANT 1 OR LIST 5
			FLD_COMP_VALUES(V_TBL_FLD_CNT).FLD_COMP_VAL := V_FLDS_REC.COMP_VALUE;
		ELSIF V_FLDS_REC.COMP_VALUE_TYPE = 2 THEN -- SCREEN BLOCK
			FLD_COMP_VALUES(V_TBL_FLD_CNT).FLD_COMP_VAL := V_FLDS_REC.COMP_VALUE;
			FLD_COMP_VALUES(V_TBL_FLD_CNT).FLD_COMP_VAL := NAME_IN(FLD_COMP_VALUES(V_TBL_FLD_CNT).FLD_COMP_VAL);
		ELSIF V_FLDS_REC.COMP_VALUE_TYPE = 3 THEN -- PARAMETER
			FLD_COMP_VALUES(V_TBL_FLD_CNT).FLD_COMP_VAL := V_FLDS_REC.COMP_VALUE;
			FLD_COMP_VALUES(V_TBL_FLD_CNT).FLD_COMP_VAL := NAME_IN(FLD_COMP_VALUES(V_TBL_FLD_CNT).FLD_COMP_VAL);

			IF 	V_FLDS_REC.COMP_OPERATOR NOT IN ('IS NULL', 'IS NOT NULL') THEN

				RAISE FORM_TRIGGER_FAILURE;
			ELSE
				FLD_COMP_VALUES(V_TBL_FLD_CNT).FLD_COMP_VAL := NULL;
			END IF;
		END IF;
	
	END FILL_TBL_VALUES;

	FUNCTION GET_WHERE_COMP_VAL(V_FLDS_REC CON_FIELDS%ROWTYPE) RETURN VARCHAR2 IS
--------------------------------------------------------------------------------    
	RET_VAL VARCHAR2(200);
	BEGIN
		IF NOT LNK.VALID_COMP_OPRT(V_FLDS_REC.COMP_OPERATOR)  THEN
			STR.DISPALERT(' ⁄—Ì› ‘—Êÿ ⁄—÷ «·ÊŸ«∆› €Ì— ’ÕÌÕ - ‰Ê⁄ „‰ÿﬁ «·„ﬁ«—‰Â €Ì— ’ÕÌÕ ·ÕﬁÊ· «·‘—ÿ','ERROR');
			RAISE FORM_TRIGGER_FAILURE;
		END IF;

		IF V_FLDS_REC.COMP_OPERATOR IN ('IS NULL', 'IS NOT NULL') THEN
			RET_VAL := NULL;
		ELSE
			IF V_FLDS_REC.COMP_VALUE_TYPE = 1 THEN -- CONSTANT
				RET_VAL  := ''''||V_FLDS_REC.COMP_VALUE||'''';
			ELSIF V_FLDS_REC.COMP_VALUE_TYPE = 5 THEN -- LIST 
				RET_VAL  := V_FLDS_REC.COMP_VALUE;
				RET_VAL := REPLACE(RET_VAL, '(');
				RET_VAL := REPLACE(RET_VAL, ')');
				RET_VAL := '('||RET_VAL||')';	
			ELSIF V_FLDS_REC.COMP_VALUE_TYPE IN (2,3) THEN -- SCREEN BLOCK 2 OR PARAMETER 3
				RET_VAL := V_FLDS_REC.COMP_VALUE; 
				RET_VAL := NAME_IN(RET_VAL);
				RET_VAL  := ''''||RET_VAL||'''';
			ELSIF V_FLDS_REC.COMP_VALUE_TYPE = 4 THEN	-- FIELD IN TABLE
				RET_VAL  := V_FLDS_REC.COMP_VALUE;				
			ELSE
				STR.DISPALERT(' ⁄—Ì› ‘—Êÿ ⁄—÷ «·ÊŸ«∆› €Ì— ’ÕÌÕ - ‰Ê⁄ „’œ— ﬁÌ„… «·„ﬁ«—‰Â ·ÕﬁÊ· «·‘—ÿ €Ì— ’ÕÌÕ ','ERROR');
				RAISE FORM_TRIGGER_FAILURE;
			END IF;
		END IF;
		
		RETURN RET_VAL;
	END GET_WHERE_COMP_VAL;

PACKAGE BODY TMPLT IS

    CURSOR CUR_Fun_Items (V_USER_ID     VARCHAR2,
                          V_MODULE_ID   NUMBER
                         )
    IS
          SELECT DISTINCT SO.SYS_OS_FILE_NAME       CD,
                          SO.SYS_ANAME              TITLE,
                          SO.FUN_MODULE             MID,
                          NULL        PATH,
                          NVL (REP_FRM_FLG, 0)      FORM_FLAG,
                          NVL (REP_PRNT_FLG, 0)     PRNT_FLAG,
                          1                         ACT
            FROM SC_SYSTEM_OBJECTS SO, SC_SYSOBJ_ROLE_PRIVS SOF
           WHERE     SO.SYSOBJ_ID = SOF.SYSOBJ_ID
                 AND SO.SYS_STATUS = 1
                 AND SO.SYS_TYPE = 7
                 AND SO.MAINSYSOBJ_ID = V_MODULE_ID
                 AND SOF.RL_SYS_ACTIVE = 1
                 AND SOF.ROLE_ID IN
                         (SELECT UR.ROLE_ID
                            FROM SC_USER_ROLES UR
                           WHERE     UR.USER_ID = V_USER_ID
                                 AND UR.PORT_CD = TMPLT.V_PORT_CD
                                 AND UR.UR_ACTIVE = 1
                          UNION
                          SELECT GR.ROLE_ID
                            FROM SC_GROUP_ROLES GR, SC_USER_GROUPS UG
                           WHERE     UG.PORT_CD = TMPLT.V_PORT_CD
                                 AND UG.USER_ID = V_USER_ID
                                 AND GR.GROUP_ID = UG.GROUP_ID
                                 AND GR.GR_ACTIVE = 1
                          MINUS
                          SELECT ROLE_ID
                            FROM SC_USER_EXCLUDED_ROLES UER
                           WHERE     UER.USER_ID = V_USER_ID
                                 AND UER.PORT_CD = TMPLT.V_PORT_CD
                                 AND UER.UER_ACTIVE = 1)
        GROUP BY SO.SYS_OS_FILE_NAME,
                 SO.SYS_ANAME,
                 SO.FUN_MODULE,
                 SO.SYS_OS_LOCATION,
                 REP_FRM_FLG,
                 REP_PRNT_FLG
        ORDER BY SO.SYS_OS_FILE_NAME DESC;

	PROCEDURE LOAD_FUNCTIONS(USER_ID VARCHAR2, MODULE_ID NUMBER) IS
------------------------------------------------------------------------------
	REC_FUN_ITEM 		TMPLT.CUR_Fun_Items%ROWTYPE;
BEGIN
		OPEN TMPLT.CUR_FUN_ITEMS(USER_ID, MODULE_ID); 
		FETCH TMPLT.CUR_FUN_ITEMS INTO REC_FUN_ITEM;
		WHILE(TMPLT.CUR_FUN_ITEMS%FOUND)
		LOOP
			TMPLT.FUN_ITEMS_COUNT := TMPLT.FUN_ITEMS_COUNT + 1;

			TMPLT.FUN_ITEMS(TMPLT.FUN_ITEMS_COUNT).TITLE 			:=  REC_FUN_ITEM.TITLE;
 			TMPLT.FUN_ITEMS(TMPLT.FUN_ITEMS_COUNT).CD 				:= REC_FUN_ITEM.CD;
 			TMPLT.FUN_ITEMS(TMPLT.FUN_ITEMS_COUNT).MID 				:= REC_FUN_ITEM.MID; 
 			TMPLT.FUN_ITEMS(TMPLT.FUN_ITEMS_COUNT).PATH 			:= REC_FUN_ITEM.PATH;
 			TMPLT.FUN_ITEMS(TMPLT.FUN_ITEMS_COUNT).FORM_FLAG 	:= REC_FUN_ITEM.FORM_FLAG;
 			TMPLT.FUN_ITEMS(TMPLT.FUN_ITEMS_COUNT).PRNT_FLAG 	:= REC_FUN_ITEM.PRNT_FLAG;
			TMPLT.FUN_ITEMS(TMPLT.FUN_ITEMS_COUNT).ACT 				:= REC_FUN_ITEM.ACT;

			FETCH TMPLT.CUR_FUN_ITEMS INTO REC_FUN_ITEM;
		END LOOP;
		CLOSE TMPLT.CUR_FUN_ITEMS;
END LOAD_FUNCTIONS;

  PROCEDURE Populate_Functions IS
------------------------------------------------------------------------------
	FN_COUNT NUMBER := 0;
	V_VIEWPORT_Y_POS NUMBER;
	V_VIEWPORT_Y_POS_ON_CANVAS NUMBER;
	V_VIEWPORT_HEIGHT NUMBER;
	ROW_NBR NUMBER;
	ROW_COUNT NUMBER := 1;
	TITLE_SEQ NUMBER := 1;
	ACT_FUNS NUMBER;
	V_POSITION NUMBER;
  V_ITEM       ITEM;
	V_X_POS          NUMBER;
	V_Y_POS          NUMBER;
	V_BTN_WIDTH  NUMBER;
	V_BTN_HEIGHT NUMBER; 
	V_WIDTH      NUMBER := 2.6;
  V_HEIGHT     NUMBER := 2.43;
	BEGIN
		IF TMPLT.FUN_ITEMS_COUNT > 0 THEN
				WHILE ROW_COUNT <= TMPLT.FUN_ITEMS_COUNT AND TITLE_SEQ <= TMPLT.MAX_FUN_ITEMS_COUNT
				LOOP
					IF TMPLT.FUN_ITEMS(ROW_COUNT).ACT = 1 THEN
						COPY(TMPLT.FUN_ITEMS(ROW_COUNT).TITLE, 'LSide.fn'||TITLE_SEQ);
						SET_ITEM_PROPERTY('LSide.fn'||TITLE_SEQ,VISIBLE, PROPERTY_TRUE);
						SET_ITEM_PROPERTY('LSide.fn'||TITLE_SEQ, VISUAL_ATTRIBUTE, 'STD_NORMAL_FUNCTION');
						TITLE_SEQ := TITLE_SEQ + 1;
					END IF;
					ROW_COUNT := ROW_COUNT + 1;
				END LOOP;
				ACT_FUNS := TITLE_SEQ - 1;
				
				/*
         Hide unused function items.
         If more than 25 active functions exist, hide from TITLE_SEQ to 50.
         If 25 or fewer active functions exist, hide from TITLE_SEQ to 25.
      */

		    IF ACT_FUNS > 25 THEN
					FOR ROW_COUNT IN TITLE_SEQ..TMPLT.MAX_FUN_ITEMS_COUNT LOOP
							SET_ITEM_PROPERTY('LSide.fn'||ROW_COUNT,VISIBLE, PROPERTY_FALSE);				
					END LOOP;
				ELSE
					TMPLT.MAX_FUN_ITEMS_COUNT := 25;
					FOR ROW_COUNT IN TITLE_SEQ..TMPLT.MAX_FUN_ITEMS_COUNT LOOP	
							SET_ITEM_PROPERTY('LSide.fn'||ROW_COUNT,VISIBLE, PROPERTY_FALSE);											
					END LOOP;
				END IF;	
				
				IF ACT_FUNS = 0 THEN
					GO_ITEM('DUMMY');
					SET_ITEM_PROPERTY('FUN_BUT', ENABLED, PROPERTY_FALSE);
				ELSE
					IF ACT_FUNS > 25 THEN
						  SET_VIEW_PROPERTY('LSIDE', VIEWPORT_X_POS, 2.815  );
						  SET_VIEW_PROPERTY('LSIDE', VIEWPORT_X_POS_ON_CANVAS,0.000);
						  SET_VIEW_PROPERTY('LSIDE', VIEWPORT_Y_POS,0.260);
						  SET_VIEW_PROPERTY('LSIDE', VIEWPORT_Y_POS_ON_CANVAS,0.000);
						  
						 
					ELSE
						  TMPLT.MAX_FUN_ITEMS_COUNT := 25;
						  V_WIDTH       := 2.6;
  						V_HEIGHT      := 2.43;
						  V_POSITION := (TMPLT.MAX_FUN_ITEMS_COUNT - ACT_FUNS)* 0.194;
							V_VIEWPORT_Y_POS := TMPLT.FUN_ITEMS_Y_POS  + V_POSITION;
							V_VIEWPORT_Y_POS := 0.26;
							V_VIEWPORT_Y_POS_ON_CANVAS := TMPLT.FUN_ITEMS_VIEW_Y_POS + V_POSITION;
						  V_VIEWPORT_HEIGHT := TMPLT.HEIGHT  - V_POSITION;
							/*SE_VIEW_PROPERTY('LSIDE', VIEWPORT_X_POS, 5.815  );
							--SET_VIEW_PROPERTY('LSIDE', VIEWPORT_X_POS, 6.5  );
						 	SET_VIEW_PROPERTY('LSIDE', VIEWPORT_Y_POS,V_VIEWPORT_Y_POS  );
							SET_VIEW_PROPERTY('LSIDE', VIEWPORT_Y_POS_ON_CANVAS, V_VIEWPORT_Y_POS_ON_CANVAS );
							*/
						 
   						--V_ITEM := FIND_ITEM('TOOLBAR.FUN_BUT');
   						V_ITEM := FIND_ITEM(NAME_IN ('SYSTEM.TRIGGER_ITEM'));
   						V_X_POS    := TO_NUMBER(GET_ITEM_PROPERTY(V_ITEM, X_POS));
   						V_Y_POS    :=  TO_NUMBER(GET_VIEW_PROPERTY('TOOLBAR', VIEWPORT_Y_POS));
   						
   						--SET_VIEW_PROPERTY ( 'LSIDE' , WIDTH , v_width) ;
   					--	SET_VIEW_PROPERTY ( 'LSIDE' , HEIGHT , v_height) ;
    					SET_VIEW_PROPERTY ( 'LSIDE' , VIEWPORT_X_POS , 8.417 - v_width ) ;
    					--SET_VIEW_PROPERTY ( 'LSIDE' , VIEWPORT_Y_POS , v_y - v_height ) ;
    					SET_VIEW_PROPERTY ( 'LSIDE' , VIEWPORT_Y_POS , V_Y_POS - (ACT_FUNS* 0.194) - 1 ) ;
   						
					END IF;	
					
						SET_VIEW_PROPERTY('LSIDE', VISIBLE, PROPERTY_TRUE);
						SET_VIEW_PROPERTY('TOOLBAR', VISIBLE, PROPERTY_TRUE);	      
				END IF;
		END IF;
	  TMPLT.MAX_FUN_ITEMS_COUNT := 50;
	END Populate_Functions;

	PROCEDURE ENB_DIS_FUN_BUT IS 
------------------------------------------------------------------------------
	ACT_FUNS NUMBER := 0;
	ROW_COUNT NUMBER := 1;
	BEGIN
		IF GET_ITEM_PROPERTY('FUN_BUT', VISIBLE) = 'TRUE' THEN
			IF TMPLT.FUN_ITEMS_COUNT > 0 THEN
						WHILE ROW_COUNT <= TMPLT.FUN_ITEMS_COUNT AND ACT_FUNS = 0 LOOP
						IF TMPLT.FUN_ITEMS(ROW_COUNT).ACT = 1 THEN
							ACT_FUNS := ACT_FUNS + 1;
						END IF;
						ROW_COUNT := ROW_COUNT + 1;
					END LOOP;
			END IF;
					
			IF ACT_FUNS = 0 THEN
				SET_ITEM_PROPERTY('FUN_BUT', ENABLED, PROPERTY_FALSE);
			ELSE
				SET_ITEM_PROPERTY('FUN_BUT', ENABLED, PROPERTY_TRUE);			
			END IF;
		END IF;
	END ENB_DIS_FUN_BUT ;

	PROCEDURE MOUSE_DOWN IS
------------------------------------------------------------------------------
		vis VARCHAR2(200);
	BEGIN
		IF (UPPER(NAME_IN('SYSTEM.MOUSE_CANVAS')) <> 'LSIDE' ) THEN
			vis :=  GET_VIEW_PROPERTY('LSIDE', visible);
			IF vis = 'TRUE' THEN
				SET_VIEW_PROPERTY('LSIDE', visible, PROPERTY_FALSE); 
				GO_ITEM(NAME_IN('PARAMETER.P_CURRENT_ITEM'));
				COPY ('' , 'PARAMETER.P_CURRENT_ITEM');
				SET_VIEW_PROPERTY('TOOLBAR', VISIBLE, PROPERTY_TRUE);

				IF TMPLT.USING_KEYB = TRUE THEN
					Set_Item_Property('LSIDE.FN'||TMPLT.CURSOR_AT, VISUAL_ATTRIBUTE, 'STD_NORMAL_FUNCTION');
					TMPLT.USING_KEYB := FALSE;
					TMPLT.CURSOR_AT := 0;
				END IF;		

			ELSE
				SET_VIEW_PROPERTY('TOOLBAR', VISIBLE, PROPERTY_TRUE);
			END IF;
		END IF;
	END MOUSE_DOWN ;

PROCEDURE KEYB_UP_DOWN(DIRECTION NUMBER) IS 
------------------------------------------------------------------------------
ACT_FUNS NUMBER := 0;
BEGIN
	IF TMPLT.USING_KEYB = TRUE THEN
		IF DIRECTION <> 0 THEN
			Set_Item_Property('LSIDE.FN'||TMPLT.CURSOR_AT, VISUAL_ATTRIBUTE, 'STD_NORMAL_FUNCTION');	
		END IF;
		IF TMPLT.FUN_ITEMS_COUNT > 0 THEN	
				FOR ROW_COUNT IN 1..TMPLT.FUN_ITEMS_COUNT LOOP
					IF TMPLT.FUN_ITEMS(ROW_COUNT).ACT = 1 THEN
						ACT_FUNS := ACT_FUNS + 1;
					END IF;
				END LOOP;
		END IF;
		
		IF ACT_FUNS > TMPLT.MAX_FUN_ITEMS_COUNT THEN
			ACT_FUNS := TMPLT.MAX_FUN_ITEMS_COUNT;
		END IF;

		TMPLT.CURSOR_AT := TMPLT.CURSOR_AT + DIRECTION;

		IF TMPLT.CURSOR_AT <= 0 THEN
			TMPLT.CURSOR_AT := ACT_FUNS;
		ELSIF TMPLT.CURSOR_AT > ACT_FUNS THEN
			TMPLT.CURSOR_AT := 1;
		END IF;
		Set_Item_Property('LSIDE.FN'||TMPLT.CURSOR_AT, VISUAL_ATTRIBUTE, 'STD_CURRENT_FUNCTION');	
	END IF;
END KEYB_UP_DOWN;

------------------------------------------------------------------------------

	ROW_COUNT NUMBER := 1;
	RET NUMBER;

	  IF CD IS NOT NULL THEN
	  	WHILE ROW_COUNT <= TMPLT.FUN_ITEMS_COUNT AND NOT FOUND_CD LOOP
	  		IF TMPLT.FUN_ITEMS(ROW_COUNT).CD LIKE CD THEN 
	  			FOUND_CD := TRUE;
	  		ELSE
		  		ROW_COUNT := ROW_COUNT + 1;
	  		END IF;
	  	END LOOP;
	  	
	  	IF FOUND_CD THEN
	  			TMPLT.FUN_ITEMS(ROW_COUNT).ACT := 0;
  				TMPLT.ENB_DIS_FUN_BUT ;	  			
	  	END IF;
	  END IF;
END DISABLE_FUN;
